        const canvas = document.getElementById('magic-canvas');

        const ctx = canvas.getContext('2d');

        function resizeCanvas() {

        class FloatingDoodle {

            constructor() {

                this.reset(true);

            }

            reset(initial = false) {

                this.x = Math.random() * canvas.width;

                this.y = initial ? Math.random() * canvas.height : canvas.height + 20;

                this.size = Math.random() * 6 + 6.5;

                this.speedY = -(Math.random() * 0.4 + 0.18);

                this.swaySpeed = Math.random() * 0.018 + 0.008;

                this.phase = Math.random() * Math.PI * 2;

                this.alpha = Math.random() * 0.35 + 0.35;

                const types = ['star', 'heart', 'bolt', 'dot'];

                this.type = types[Math.floor(Math.random() * types.length)];

                this.color = Math.random() > 0.35 ? '#fde08d' : '#f4a28c';

            }

            update() {

                this.y += this.speedY;

                this.x += Math.sin(this.phase) * 0.3;

                this.phase += this.swaySpeed;

                if (this.y < -25) this.reset();

            }

            draw() {

                ctx.save();

                ctx.translate(this.x, this.y);

                ctx.rotate(Math.sin(this.phase) * 0.25);

                ctx.globalAlpha = this.alpha;

                ctx.strokeStyle = this.color;

                ctx.fillStyle = this.color;

                ctx.lineWidth = 1.6;

                ctx.lineCap = 'round';

                ctx.lineJoin = 'round';


                const s = this.size;

                if (this.type === 'star') {

                    ctx.beginPath();

                    ctx.moveTo(0, -s);

                    ctx.quadraticCurveTo(0, 0, s, 0);

                    ctx.quadraticCurveTo(0, 0, 0, s);

                    ctx.quadraticCurveTo(0, 0, -s, 0);

                    ctx.quadraticCurveTo(0, 0, 0, -s);

                    ctx.stroke();

                } else if (this.type === 'heart') {

                    ctx.beginPath();

                    ctx.moveTo(0, s * 0.3);

                    ctx.bezierCurveTo(-s * 0.6, -s * 0.4, -s * 1.1, s * 0.2, 0, s * 0.9);

                    ctx.bezierCurveTo(s * 1.1, s * 0.2, s * 0.6, -s * 0.4, 0, s * 0.3);

                    ctx.stroke();

                } else if (this.type === 'bolt') {

                    ctx.beginPath();

                    ctx.moveTo(s * 0.2, -s * 0.7);

                    ctx.lineTo(-s * 0.3, 0);

                    ctx.lineTo(s * 0.2, 0);

                    ctx.lineTo(-s * 0.2, s * 0.7);

                    ctx.stroke();

                } else {

                    ctx.beginPath();

                    ctx.arc(0, 0, s * 0.3, 0, Math.PI * 2);

                    ctx.fill();

                }

                ctx.restore();

            }

        }


        let animationStarted = false;

        function startMagicBackground() {

            if (animationStarted) return;

            animationStarted = true;

            resizeCanvas();

            const count = Math.max(22, Math.min(35, Math.floor(window.innerWidth / 12)));

            doodles = Array.from({ length: count }, () => new FloatingDoodle());

            

            function animateDoodles() {

                ctx.clearRect(0, 0, canvas.width, canvas.height);

                doodles.forEach(d => {

                    d.update();

                    d.draw();

                });

                requestAnimationFrame(animateDoodles);

            }

            animateDoodles();

        }


        if (document.readyState === 'loading') {

            document.addEventListener('DOMContentLoaded', startMagicBackground);

        } else {

            startMagicBackground();

        }

        window.addEventListener('load', startMagicBackground);
